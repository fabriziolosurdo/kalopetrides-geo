  window.KL_I18N_EL = EL; // allow per-page dictionaries to extend (Object.assign)

  function getLang() { return localStorage.getItem(STORE) === "el" ? "el" : "en"; }

  function setHTML(el, val) {
    if (el.getAttribute("data-i18n-o") === null) el.setAttribute("data-i18n-o", el.innerHTML);
    el.innerHTML = val;
  }
  function restoreHTML(el) {
    var o = el.getAttribute("data-i18n-o");
    if (o !== null) el.innerHTML = o;
  }

  function apply(lang) {
    var dict = lang === "el" ? window.KL_I18N_EL : null;

    document.documentElement.setAttribute("lang", lang);
    document.documentElement.setAttribute("data-lang", lang);

    document.querySelectorAll("[data-i18n]").forEach(function (el) {
      var k = el.getAttribute("data-i18n");
      if (dict && dict[k] != null) setHTML(el, dict[k]);
      else restoreHTML(el);
    });

    document.querySelectorAll("[data-i18n-ph]").forEach(function (el) {
      var k = el.getAttribute("data-i18n-ph");
      if (el.getAttribute("data-i18n-pho") === null) el.setAttribute("data-i18n-pho", el.getAttribute("placeholder") || "");
      el.setAttribute("placeholder", dict && dict[k] != null ? dict[k] : el.getAttribute("data-i18n-pho"));
    });

    document.querySelectorAll("[data-i18n-aria]").forEach(function (el) {
      var k = el.getAttribute("data-i18n-aria");
      if (el.getAttribute("data-i18n-ariao") === null) el.setAttribute("data-i18n-ariao", el.getAttribute("aria-label") || "");
      el.setAttribute("aria-label", dict && dict[k] != null ? dict[k] : el.getAttribute("data-i18n-ariao"));
    });

    // <title>
    var file = (location.pathname.split("/").pop() || "index.html") || "index.html";
    if (lang === "el" && TITLES[file]) {
      if (!window.__klTitleOrig) window.__klTitleOrig = document.title;
      document.title = TITLES[file];
    } else if (window.__klTitleOrig) {
      document.title = window.__klTitleOrig;
    }

    // toggle buttons state
    document.querySelectorAll(".lang-btn").forEach(function (b) {
      var on = b.getAttribute("data-lang") === lang;
      b.classList.toggle("is-active", on);
      b.setAttribute("aria-pressed", on ? "true" : "false");
    });

    localStorage.setItem(STORE, lang);
  }
